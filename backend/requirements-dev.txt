# choyeon-panel 后端「测试/开发」依赖（生产环境不需要，别写进 requirements.txt）
#
# 为什么单独一个文件：
# - requirements.txt 与 pyproject.toml 的 [project].dependencies 一致，
#   被 scripts/common.sh 的 venv_create() 用于生产安装，装测试包会污染线上环境。
# - starlette>=1.2（本作装到 1.7.0）的 TestClient 硬依赖 httpx2：
#   干净环境里 `from starlette.testclient import TestClient` 直接
#   RuntimeError（"requires the httpx2 package"），只有本地 venv 恰好带着
#   旧 httpx 时才退化成 deprecation warning。所以本地绿 ≠ CI 绿，
#   CI 必须显式安装本文件（见 .github/workflows/ci.yml 的「安装依赖」步骤）。
httpx2>=2.0
# 与 CI 同一套静态检查（.github/workflows/ci.yml 原来在这里之外内联 `pip install ruff mypy`，
# 于是 CI 与开发机是两套工具集，"本地复现 CI"没有落点；现在两边只读这一份清单）
ruff>=0.17
mypy>=1.11
# 依赖漏洞审计（ci.yml 的后端 job 用它审 requirements.txt，阻断项）。
# 审计对象是 requirements.txt 里那份范围声明解析出来的版本——那正是新机器装上去的东西。
pip-audit>=2.7
